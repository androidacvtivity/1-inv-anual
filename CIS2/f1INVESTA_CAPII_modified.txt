function f1INVESTA_CAPII() {

    function el(idMd, rind, col) {
        return $("#67_1140_" + idMd + "_" + rind + "_" + col);
    }

    function n($el) {
        var value = $.trim($el.val());

        if (value === "") {
            return 0;
        }

        value = value.replace(/\s/g, "").replace(",", ".");

        var result = parseFloat(value);
        return isNaN(result) ? 0 : result;
    }

    function set1($el, value) {
        $el.val(Number(value).toFixed(1));
    }

    function setReadonly($el) {
        $el.prop("readonly", true);
    }

    function calcCol7(row) {
        return n(el(row.idMd, row.rind, 2)) +
            n(el(row.idMd, row.rind, 3)) -
            n(el(row.idMd, row.rind, 4)) +
            n(el(row.idMd, row.rind, 5)) +
            n(el(row.idMd, row.rind, 6));
    }

    function calcGroup(totalRow, detailRows) {

        // Total row: C2-C7 readonly
        for (var col = 2; col <= 7; col++) {
            setReadonly(el(totalRow.idMd, totalRow.rind, col));
        }

        // Detail rows: C7 readonly
        detailRows.forEach(function (row) {
            setReadonly(el(row.idMd, row.rind, 7));
        });

        // Total row C2-C6 = sum of detail rows C2-C6
        for (var c = 2; c <= 6; c++) {
            var sum = 0;

            detailRows.forEach(function (row) {
                sum += n(el(row.idMd, row.rind, c));
            });

            set1(el(totalRow.idMd, totalRow.rind, c), sum);
        }

        // Detail rows C7 = C2 + C3 - C4 + C5 + C6
        detailRows.forEach(function (row) {
            set1(el(row.idMd, row.rind, 7), calcCol7(row));
        });

        // Total row C7 = C2 + C3 - C4 + C5 + C6
        set1(el(totalRow.idMd, totalRow.rind, 7), calcCol7(totalRow));
    }

    var total130 = { idMd: "62688", rind: "130" };
    var rows130 = [

    var total140 = { idMd: "62699", rind: "140" };
